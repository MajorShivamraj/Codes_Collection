def create_questionpaper():

    print("!!! It is prefered if you make a Question Paper with 10Q or More....!!!")
    q=int(input("Number of Questions = "))
    o=int(input("By Default number of opptions = "))
    questions=[]
    for i in range(0,q):
        question=[]#here this list is inside the for loop because it is getting empty everytime the list is appended in the main list......
        for j in range(0,o+2):
            pip=["Q :- ","Option 1:","Option 2:","Option 3:","Option 4:","Answer = "]#here the pip is helping to get the Q, option1, option2.....
            ques=str(input(f"Enter {pip[j]}"))
            question.append(ques)
        print(f"Question {i+1}: addded succesfully !!!")
        questions.append(question)
    print(questions)
