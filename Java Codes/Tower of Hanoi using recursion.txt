// Tower of Henoi Solver...
public class Main {
    public static int i=0;
    public static void tower_of_henoi(int n,String src, String help, String dest){
        i++;
        if(n==1){
            System.out.println("Transfer disk "+ n + " from Tower "+src+" To Tower "+dest);
            return;
        }
        tower_of_henoi(n-1,src,dest,help);
        System.out.println("Transfer disk "+ n + " from Tower "+src+" To Tower "+dest);
        tower_of_henoi(n-1,help,src,dest);

    }
    public static void main(String[] args) {
        int n=1;
        tower_of_henoi(4,"S","H","D");
        System.out.println("It took total "+ i+ " Steps");
    }
}
